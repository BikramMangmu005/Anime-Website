  },
  {

    title: "Nature & Life",
    image: "https://images.unsplash.com/photo-1464822759023-fed622ff2c3b?q=80&w=1000&auto=format&fit=crop",

  },

export default function SnapStoryHero() {
  const [currentIndex, setCurrentIndex] = useState(0);

  const active = slides[currentIndex];

  const handlePrev = () => {
    setCurrentIndex((prev) => (prev === 0 ? slides.length - 1 : prev - 1));
  };

  const handleNext = () => {
    setCurrentIndex((prev) => (prev === slides.length - 1 ? 0 : prev + 1));
  };

  return (
    <div className="relative min-h-screen w-full overflow-hidden bg-[#0d0a0a] text-white font-sans selection:bg-white/20">
      
      {/* Background Image with Dark Vignette/Blur */}
      <div
        className="absolute inset-0 bg-cover bg-center bg-no-repeat opacity-40 blur-[2px] scale-105 transition-all duration-1000 ease-out"
        style={{ backgroundImage: `url("${active.bg}")` }}
      />

      {/* Heavy Cinematic Gradient Overlays */}
      <div className="absolute inset-0 bg-gradient-to-r from-[#0d0a0a] via-[#0d0a0a]/80 to-transparent" />
      <div className="absolute inset-0 bg-gradient-to-t from-[#0d0a0a] via-transparent to-[#0d0a0a]/60" />

      {/* Main Container */}
      <div className="relative z-10 flex min-h-screen flex-col justify-between px-6 md:px-12 lg:px-20 py-6">
        
        {/* Navigation Header */}
        <header className="flex items-center justify-between">
          <h1 className="text-xl font-semibold tracking-wide font-serif text-white/90">
            SnapStory
          </h1>

          <nav className="hidden md:flex items-center space-x-10 text-sm font-medium text-white/70">
            {["Discover", "Styles", "Contacts", "Login"].map((item) => (
              <a
                key={item}
                href={`#${item.toLowerCase()}`}
                className="hover:text-white transition-colors duration-200"
              >
                {item}
              </a>
            ))}
          </nav>
        </header>

        {/* Hero Body Section */}
        <main className="grid grid-cols-1 lg:grid-cols-12 gap-8 items-center my-auto py-8 mt-50">
          
          {/* Left Column: Typography & Description */}
          <div className="lg:col-span-5 space-y-6">
            <h2 className="text-5xl sm:text-6xl lg:text-7xl font-serif tracking-wide text-white drop-shadow-md transition-all duration-700">
              {active.title}
            </h2>

            <p className="text-white/70 text-xs sm:text-sm leading-relaxed max-w-md font-light transition-all duration-700">
              {active.description}
            </p>

            <div className="pt-2">
              <button className="rounded-full bg-[#2a2222]/80 hover:bg-[#382e2e] border border-white/10 px-7 py-3 text-xs tracking-wider uppercase font-medium text-white backdrop-blur-md transition-all duration-300 shadow-lg cursor-pointer">
                Explore now
              </button>
            </div>
          </div>

          {/* Right Column: Sliding Track with Card-Side Overlay Buttons */}
          <div className="lg:col-span-7 relative flex items-center py-6">
            
            {/* Left Arrow Button */}
            <button
              onClick={handlePrev}
              aria-label="Previous slide"
              className="absolute left-0 z-40 -ml-5 h-12 w-12 rounded-full bg-black/60 hover:bg-black/80 border border-white/20 flex items-center justify-center text-white backdrop-blur-md transition-all duration-200 cursor-pointer shadow-xl"
            >
              <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={2} stroke="currentColor" className="w-5 h-5">
                <path strokeLinecap="round" strokeLinejoin="round" d="M15.75 19.5L8.25 12l7.5-7.5" />
              </svg>
            </button>

            {/* Slider Track Container */}
            <div className="w-full overflow-hidden px-4">
              <div 
                className="flex items-center gap-6 transition-transform duration-700 ease-in-out"
                style={{ transform: `translateX(calc(-${currentIndex} * (340px + 24px)))` }}
              >
                {slides.map((slide, index) => {
                  const isCurrent = index === currentIndex;

                  return (
                    <div
                      key={slide.id}
                      onClick={() => setCurrentIndex(index)}
                      className={`group relative flex-shrink-0 cursor-pointer overflow-hidden rounded-3xl h-[400px] w-[340px] transition-all duration-700 ease-in-out ${
                        isCurrent
                          ? "border-2 border-white shadow-[0_25px_60px_-15px_rgba(0,0,0,0.9)] opacity-100 scale-100"
                          : "border border-white/10 opacity-50 scale-95 hover:opacity-75"
                      }`}
                    >
                      <img
                        src={slide.image}
                        alt={slide.title}
                        className="h-full w-full object-cover transition-transform duration-700 group-hover:scale-105 pointer-events-none"
                      />

                      {/* Inner Card Gradient Overlay */}
                      <div className="absolute inset-0 bg-gradient-to-t from-black/80 via-transparent to-transparent" />
                    </div>
                  );
                })}
              </div>
            </div>

            {/* Right Arrow Button */}
            <button
              onClick={handleNext}
              aria-label="Next slide"
              className="absolute right-0 z-40 -mr-5 h-12 w-12 rounded-full bg-black/60 hover:bg-black/80 border border-white/20 flex items-center justify-center text-white backdrop-blur-md transition-all duration-200 cursor-pointer shadow-xl"
            >
              <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" strokeWidth={2} stroke="currentColor" className="w-5 h-5">
                <path strokeLinecap="round" strokeLinejoin="round" d="M8.25 4.5l7.5 7.5-7.5 7.5" />
              </svg>
            </button>

          </div>
        </main>

        {/* Footer Slider Pagination Indicators */}
        <footer className="flex items-center justify-center pb-2">
          <div className="flex space-x-2 bg-black/40 backdrop-blur-md px-4 py-2 rounded-full border border-white/5">
            {slides.map((_, idx) => (
              <button
                key={idx}
                onClick={() => setCurrentIndex(idx)}
                aria-label={`Go to slide ${idx + 1}`}
                className={`h-2 rounded-full cursor-pointer transition-all duration-300 ${
                  currentIndex === idx
                    ? "w-6 bg-white"
                    : "w-2 bg-white/40 hover:bg-white/70"
                }`}
              />
            ))}
          </div>
        </footer>

      </div>
    </div>
  );
}
